    <script>
        class SoundFX {
            constructor() {
                this.ctx = null;
            }

            init() {
                if (!this.ctx) {
                    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
                }
            }

            playBeep(freq = 800, type = 'sine', duration = 0.1) {
                try {
                    this.init();
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = type;
                    osc.frequency.value = freq;
                    gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + duration);
                } catch(e){}
            }

            playScan() {
                this.playBeep(1000, 'sine', 0.05);
                setTimeout(() => this.playBeep(1400, 'sine', 0.05), 80);
            }

            playSuccess() {
                try {
                    this.init();
                    const notes = [523.25, 659.25, 783.99, 1046.50]; // C5, E5, G5, C6
                    notes.forEach((freq, idx) => {
                        setTimeout(() => {
                            this.playBeep(freq, 'triangle', 0.25);
                        }, idx * 90);
                    });
                } catch(e){}
            }

            playError() {
                try {
                    this.init();
                    this.playBeep(180, 'sawtooth', 0.25);
                    setTimeout(() => this.playBeep(130, 'sawtooth', 0.3), 110);
                } catch(e){}
            }
        }

        const sfx = new SoundFX();
    </script>
